<script>
  let timeLeft = 50;
  const total = 50;
  const timerEl = document.getElementById('timer');
  const barEl = document.getElementById('bar');
  const titleEl = document.getElementById('title');
  const msgEl = document.getElementById('msg');
  const iconEl = document.getElementById('icon');

  // Небольшой "глитч" для атмосферы
  const glitch = setInterval(() => {
    if (Math.random() < 0.3) {
      document.body.style.background = Math.random() < 0.5 ? '#0a0a0a' : '#001100';
    }
  }, 150);

  const interval = setInterval(() => {
    timeLeft--;
    timerEl.textContent = timeLeft;
    barEl.style.width = ((total - timeLeft) / total * 100) + '%';

    if (timeLeft <= 10) timerEl.style.color = '#ff0044';

    if (timeLeft <= 0) {
      clearInterval(interval);
      clearInterval(glitch);
      document.body.style.background = '#0a0a0a';
      iconEl.textContent = '✅';
      iconEl.style.animation = 'none';
      titleEl.textContent = 'РАЗБЛОКИРОВАНО';
      titleEl.classList.add('unlocked');
      timerEl.textContent = '0';
      timerEl.style.color = '#00ff88';
      msgEl.innerHTML = 'Шутка! <br>Ничего не заблокировано.';
    }
  }, 1000);
</script>
